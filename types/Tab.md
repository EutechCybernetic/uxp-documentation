# Tab








```tsx
interface Tab {
    /**
     * id of the tab
     */
    id: string,
    /**
     * label of the tab
     */
    label: string | React.ReactNode,
    /**
     * tab icon
     */
    icon?: string,
    /**
     * tab content
     */
    content: React.ReactNode
}
```

## Usage



```tsx
import {Tab} from 'uxp/components';
```
